	"github.com/google/uuid"

	if testing.Short() {
		t.Skip("Skipping repository tests in short mode.")
	}

	repo := repositories.New{{Resources}}PostgresRepository(testDB)

func seed{{Resource}}(t *testing.T, repo domain.{{Resources}}Repository) *domain.{{Resource}} {
	created, err := repo.Create(context.Background(), domain.{{Resource}}{})
	require.NoError(t, err)
	require.NotNil(t, created)

	return created
}

func Test{{Resources}}PostgresRepository_Create(t *testing.T) {
	repo := setup{{Resources}}RepoTest(t)

	t.Run("success", func(t *testing.T) {
		created := seed{{Resource}}(t, repo)

		assert.NotEmpty(t, created.UUID)
		assert.NotZero(t, created.CreatedAt)
	})
}

func Test{{Resources}}PostgresRepository_FindByUUID(t *testing.T) {
	repo := setup{{Resources}}RepoTest(t)

	t.Run("success", func(t *testing.T) {
		created := seed{{Resource}}(t, repo)

		found, err := repo.FindByUUID(context.Background(), created.UUID)

		require.NoError(t, err)
		assert.Equal(t, created, found)
	})

	t.Run("error - not found", func(t *testing.T) {
		found, err := repo.FindByUUID(context.Background(), uuid.New().String())

		var notFoundErr *domain.NotFoundError
		assert.ErrorAs(t, err, &notFoundErr)
		assert.Nil(t, found)
	})

	t.Run("error - invalid uuid", func(t *testing.T) {
		found, err := repo.FindByUUID(context.Background(), "invalid-uuid")

		var validationErr *domain.ValidationError
		assert.ErrorAs(t, err, &validationErr)
		assert.Nil(t, found)
	})
}

func Test{{Resources}}PostgresRepository_Update(t *testing.T) {
	repo := setup{{Resources}}RepoTest(t)

	t.Run("success", func(t *testing.T) {
		created := seed{{Resource}}(t, repo)

		updated, err := repo.Update(context.Background(), *created)

		require.NoError(t, err)
		assert.Equal(t, created.UUID, updated.UUID)
	})

	t.Run("error - not found", func(t *testing.T) {
		updated, err := repo.Update(context.Background(), domain.{{Resource}}{UUID: uuid.New().String()})

		var notFoundErr *domain.NotFoundError
		assert.ErrorAs(t, err, &notFoundErr)
		assert.Nil(t, updated)
	})
}

func Test{{Resources}}PostgresRepository_Delete(t *testing.T) {
	repo := setup{{Resources}}RepoTest(t)

	t.Run("success", func(t *testing.T) {
		created := seed{{Resource}}(t, repo)

		err := repo.Delete(context.Background(), created.UUID)
		require.NoError(t, err)

		found, err := repo.FindByUUID(context.Background(), created.UUID)
		var notFoundErr *domain.NotFoundError
		assert.ErrorAs(t, err, &notFoundErr)
		assert.Nil(t, found)
	})
}
