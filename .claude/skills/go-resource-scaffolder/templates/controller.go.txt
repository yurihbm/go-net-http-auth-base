package controllers

import (
	"net/http"

	"go-net-http-auth-base/internal/api"
	"go-net-http-auth-base/internal/domain"
	"go-net-http-auth-base/internal/middlewares"
)

type {{Resources}}Controller struct {
	service        domain.{{Resources}}Service
	authMiddleware middlewares.HandlerMiddleware
}

var _ Controller = (*{{Resources}}Controller)(nil)

func New{{Resources}}Controller(
	service domain.{{Resources}}Service,
	authMiddleware middlewares.HandlerMiddleware,
) *{{Resources}}Controller {
	return &{{Resources}}Controller{
		service:        service,
		authMiddleware: authMiddleware,
	}
}

func (c *{{Resources}}Controller) RegisterRoutes(router *http.ServeMux) {
	router.HandleFunc("POST /{{resources}}", c.authMiddleware.Use(c.Create))
	router.HandleFunc("GET /{{resources}}/{uuid}", c.authMiddleware.Use(c.GetByUUID))
	router.HandleFunc("PATCH /{{resources}}/{uuid}", c.authMiddleware.Use(c.Update))
	router.HandleFunc("DELETE /{{resources}}/{uuid}", c.authMiddleware.Use(c.Delete))
}

// @Summary		Create a {{resource}}
// @Description	Requires the `access_token` cookie set by POST /auth/login.
// @Tags			{{resources}}
// @Accept			json
// @Produce		json
// @Param			body	body		domain.Create{{Resource}}DTO				true	"{{Resource}} to create"
// @Success		201		{object}	api.ResponseBody[domain.{{Resource}}]	"{{Resource}} created"
// @Failure		400		{object}	api.ResponseBody[any]				"Validation error"
// @Failure		401		{object}	api.ResponseBody[any]				"Unauthorized"
// @Failure		409		{object}	api.ResponseBody[any]				"Conflict"
// @Failure		500		{object}	api.ResponseBody[any]				"Internal server error"
// @Router			/{{resources}} [post]
func (c *{{Resources}}Controller) Create(w http.ResponseWriter, r *http.Request) {
	dto, err := api.DecodeAndValidate[domain.Create{{Resource}}DTO](r)
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	created, err := c.service.Create(r.Context(), dto)
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	api.WriteJSONResponse(w, http.StatusCreated, api.ResponseBody[domain.{{Resource}}]{
		Data:    *created,
		Message: "{{resource}}.create.success",
	})
}

// @Summary		Get a {{resource}} by UUID
// @Description	Requires the `access_token` cookie set by POST /auth/login.
// @Tags			{{resources}}
// @Produce		json
// @Param			uuid	path		string								true	"{{Resource}} UUID"
// @Success		200		{object}	api.ResponseBody[domain.{{Resource}}]	"{{Resource}}"
// @Failure		400		{object}	api.ResponseBody[any]				"Validation error"
// @Failure		401		{object}	api.ResponseBody[any]				"Unauthorized"
// @Failure		404		{object}	api.ResponseBody[any]				"Not found"
// @Failure		500		{object}	api.ResponseBody[any]				"Internal server error"
// @Router			/{{resources}}/{uuid} [get]
func (c *{{Resources}}Controller) GetByUUID(w http.ResponseWriter, r *http.Request) {
	uuid, err := pathUUID(r, "uuid", "{{resources}}.invalidUUID")
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	entity, err := c.service.GetByUUID(r.Context(), uuid)
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	api.WriteJSONResponse(w, http.StatusOK, api.ResponseBody[domain.{{Resource}}]{
		Data:    *entity,
		Message: "{{resource}}.get.success",
	})
}

// @Summary		Update a {{resource}}
// @Description	Partially updates a {{resource}}. Requires the `access_token` cookie set by POST /auth/login.
// @Tags			{{resources}}
// @Accept			json
// @Produce		json
// @Param			uuid	path		string								true	"{{Resource}} UUID"
// @Param			body	body		domain.{{Resource}}UpdateDTO			true	"Fields to update"
// @Success		200		{object}	api.ResponseBody[domain.{{Resource}}]	"{{Resource}} updated"
// @Failure		400		{object}	api.ResponseBody[any]				"Validation error"
// @Failure		401		{object}	api.ResponseBody[any]				"Unauthorized"
// @Failure		404		{object}	api.ResponseBody[any]				"Not found"
// @Failure		409		{object}	api.ResponseBody[any]				"Conflict"
// @Failure		500		{object}	api.ResponseBody[any]				"Internal server error"
// @Router			/{{resources}}/{uuid} [patch]
func (c *{{Resources}}Controller) Update(w http.ResponseWriter, r *http.Request) {
	uuid, err := pathUUID(r, "uuid", "{{resources}}.invalidUUID")
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	dto, err := api.DecodeAndValidate[domain.{{Resource}}UpdateDTO](r)
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	updated, err := c.service.Update(r.Context(), uuid, dto)
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	api.WriteJSONResponse(w, http.StatusOK, api.ResponseBody[domain.{{Resource}}]{
		Data:    *updated,
		Message: "{{resource}}.update.success",
	})
}

// @Summary		Delete a {{resource}}
// @Description	Requires the `access_token` cookie set by POST /auth/login.
// @Tags			{{resources}}
// @Produce		json
// @Param			uuid	path	string	true	"{{Resource}} UUID"
// @Success		204		"{{Resource}} deleted"
// @Failure		400		{object}	api.ResponseBody[any]	"Validation error"
// @Failure		401		{object}	api.ResponseBody[any]	"Unauthorized"
// @Failure		404		{object}	api.ResponseBody[any]	"Not found"
// @Failure		500		{object}	api.ResponseBody[any]	"Internal server error"
// @Router			/{{resources}}/{uuid} [delete]
func (c *{{Resources}}Controller) Delete(w http.ResponseWriter, r *http.Request) {
	uuid, err := pathUUID(r, "uuid", "{{resources}}.invalidUUID")
	if err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	if err := c.service.Delete(r.Context(), uuid); err != nil {
		api.HandleError(r.Context(), w, err)
		return
	}

	w.WriteHeader(http.StatusNoContent)
}
